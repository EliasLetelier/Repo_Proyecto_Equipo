import pandas as pd import altair as alt # df is assumed to be already loaded and cleaned from previous steps. # If not, this block ensures it's loaded with necessary preprocessing. if 'df' not in locals() and 'df' not in globals(): df = pd.read_csv('Estadisticas U-COLOCOLO - Hoja 1.csv', skiprows=1) df.columns = [c.strip() for c in df.columns] df['Árbitro'] = df['Árbitro'].str.strip() df['Ganador'] = df['Ganador'].str.strip() top_arbitros = df['Árbitro'].value_counts().nlargest(12).index df_filtered = df[df['Árbitro'].isin(top_arbitros)].copy() viz_data = df_filtered.groupby(['Árbitro', 'Ganador']).size().reset_index(name='Cantidad') chart = alt.Chart(viz_data).mark_bar().encode( y=alt.Y('Árbitro:N', title='Árbitro', sort='-x'), x=alt.X('Cantidad:Q', title='Número de Partidos'), color=alt.Color('Ganador:N', scale=alt.Scale(domain=['U de Chile', 'Colo-Colo', 'Empate'], range=['#003399', '#000000', '#808080']), title="Resultado"), order=alt.Order('Ganador:N', sort='ascending'), tooltip=['Árbitro', 'Ganador', 'Cantidad'] ).properties( width=600, height=450, title='Historial por Árbitro: Victorias y Empates' ).configure_title( fontSize=18, anchor='start' ) chart.save('comparativo_completo_arbitros.html') chart
